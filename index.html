🔄 Live Sync — Google Sheets
Connect your master registration Google Sheet for one-click updates
One-time setup — Google Apps Script (~3 mins)
1. Open your MMI Master Registration Google Sheet
2. Click Extensions → Apps Script
3. Delete all existing code and paste the LOME script below
4. Click Deploy → New deployment → Web app
5. Set Execute as: Me  ·  Who has access: Anyone → Deploy
6. Copy the Web app URL and paste it below
⚠ Open your sheet normally at docs.google.com/spreadsheets — not the published /e/ URL — then go to Extensions → Apps Script
Runs on Google's own servers — no browser CORS issues. Your sheet stays private in Google Drive. This script works with your actual Google Sheet — not the published CSV URL.
📋 Paste this into Extensions → Apps Script 
function doGet(e) {
  // LOME Sync Script — reads MMI Master Registration List
  // Searches all sheet tabs for "Full List" or uses the first sheet
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tab = null;
  var sheets = ss.getSheets();
  // Try to find "Full List" tab (case-insensitive)
  for(var s=0; s= 0){
      tab = sheets[s]; break;
    }
  }
  if(!tab) tab = sheets[0]; // fallback to first sheet

  var data = tab.getDataRange().getValues();
  if(data.length < 2) return ContentService.createTextOutput(JSON.stringify({total:0,vip:0,std:0,ql:0,vipPct:38,weekly:{},updated:new Date().toISOString()})).setMimeType(ContentService.MimeType.JSON);

  var headers = data[0].map(function(h){ return String(h).toLowerCase().trim(); });
  // Find key columns — flexible matching
  var lsCol   = -1, dateCol = -1, typeCol = -1;
  headers.forEach(function(h,i){
    if(h.includes("lead source") || h === "ls") lsCol = i;
    if((h.includes("date") && h.includes("purchase")) || h === "date of purchase") dateCol = i;
    if(h === "tc" || h === "ticket category" || h === "type") typeCol = i;
  });
  // Fallback: use first column that contains "date"
  if(dateCol < 0) headers.forEach(function(h,i){ if(h.includes("date") && dateCol<0) dateCol=i; });

  var total=0, vip=0, std=0, ql=0;
  var weekly = {};
  for(var i=1; i=0   ? String(row[lsCol]  ||"").toLowerCase().trim() : "";
    var type = typeCol>=0 ? String(row[typeCol] ||"").toLowerCase().trim() : "";
    if(!ls && !type) continue;
    if(ls.includes("refund")) continue;
    total++;
    // VIP detection: check lead source or ticket type column
    var isVip = ls.includes("vip") || type === "vip";
    if(isVip) vip++; else std++;
    // QL detection
    if(ls.indexOf("[ql]") === 0 || ls.includes("ql buyer") || ls === "ql") ql++;
    // Weekly velocity from purchase date
    if(dateCol>=0 && row[dateCol]){
      try{
        var d = new Date(row[dateCol]);
        if(!isNaN(d.getTime())){
          var yr = d.getFullYear();
          var startOfYear = new Date(yr,0,1);
          var wk = Math.ceil((((d - startOfYear)/86400000) + startOfYear.getDay() + 1) / 7);
          var key = yr + "-W" + ("0"+wk).slice(-2);
          weekly[key] = (weekly[key]||0) + 1;
        }
      } catch(ex){}
    }
  }
  var result = {
    total:   total,
    vip:     vip,
    std:     std,
    ql:      ql,
    vipPct:  total > 0 ? Math.round(vip/total*100) : 38,
    weekly:  weekly,
    sheet:   tab.getName(),
    updated: new Date().toISOString()
  };
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
