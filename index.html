function doGet(e) {
  // LOME Sync Script v4 — counts registrations directly from Full List (Ticket
  // Quantity summed, filtered by Ticket Category + status columns — same rigor
  // as the Dashboard's own SUMIFS formulas) rather than trusting the Dashboard
  // tab's cached subtotal cells, which inherit a template typo (VIP-Complimentary
  // formula filters for ''Complementary'' and never matches real ''Complimentary''
  // rows). We don't touch the shared template — this sidesteps the bug entirely.
  // Dashboard is only read for the manually-set Target and as a cross-check.
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();

  var dashTab = findSheetByKeyword(sheets, ['dashboard']);
  var listTab = findSheetByKeyword(sheets, ['full list', 'master', 'registration', 'attendee']) || sheets[0];

  var listResult = readFullList(listTab);
  var dash = dashTab ? readDashboard(dashTab) : null;

  var total = listResult.total;
  var vip   = listResult.vip;
  var std   = listResult.std;
  var ql    = listResult.ql;

  return outputJson({
    total: total,
    vip: vip,
    std: std,
    ql: ql,
    vipPct: total > 0 ? Math.round((vip / total) * 100) : 38,
    thaiPct: listResult.total > 0 ? Math.round((listResult.thai / listResult.total) * 100) : 0,
    target: dash ? dash.target : null,
    weekly: listResult.weekly,
    source: 'fulllist',
    dashboardTotal: dash ? dash.total : null,
    mismatch: !!(dash && dash.total > 0 && dash.total !== total),
    sheet: listTab.getName(),
    updated: new Date().toISOString()
  });
}

function findSheetByKeyword(sheets, keywords) {
  for (var s = 0; s < sheets.length; s++) {
    var name = sheets[s].getName().toLowerCase();
    for (var k = 0; k < keywords.length; k++) {
      if (name.indexOf(keywords[k]) >= 0) return sheets[s];
    }
  }
  return null;
}

// Reads the Dashboard tab's own formula outputs — already computed via SUMIFS
// on Ticket Quantity, filtered by Ticket Category + status. This is the number
// the team already treats as official, and it correctly counts multi-seat rows
// (Ticket Quantity > 1) that a raw row-count would miss.
function readDashboard(tab) {
  try {
    var total = numOrNull(tab.getRange('B3').getValue());
    if (total === null) return null;
    return {
      total: total,
      target: numOrNull(tab.getRange('C3').getValue()),
      std: numOrNull(tab.getRange('G9').getValue()) || 0,
      vip: numOrNull(tab.getRange('G13').getValue()) || 0,
      ql: numOrNull(tab.getRange('F17').getValue()) || 0
    };
  } catch (err) {
    return null;
  }
}

function numOrNull(v) {
  return (typeof v === 'number' && !isNaN(v)) ? v : null;
}

// Same Ticket Category / status whitelist the Dashboard's SUMIFS formulas use,
// so weekly velocity (which the Dashboard doesn't summarize) stays consistent
// with the headline total instead of a separate fuzzy heuristic.
var VALID_TC = ['standard', 'vip'];
var VALID_STATUS = ['paid', 'guest', 'complimentary', 'ql', 'rollover'];

function readFullList(tab) {
  var data = tab.getDataRange().getValues();
  if (!data || data.length < 2) {
    return { total:0, vip:0, std:0, ql:0, thai:0, weekly:{} };
  }

  var headerRowIndex = findHeaderRow(data);
  var headers = data[headerRowIndex].map(function(h){ return String(h || '').toLowerCase().trim(); });

  var qtyCol    = findCol(headers, ['ticket quantity', 'quantity', 'qty']);
  var tcCol     = findCol(headers, ['tc', 'ticket category']);
  var statusCol = findCol(headers, ['paid/complimentary/guest/ql', 'status', 'payment status', 'order status']);
  var dateCol   = findCol(headers, ['date of purchase', 'purchase date', 'date purchased', 'created date', 'created at', 'date']);

  var total = 0, vip = 0, std = 0, ql = 0, thai = 0;
  var weekly = {};

  for (var i = headerRowIndex + 1; i < data.length; i++) {
    var row = data[i];
    var tc = tcCol >= 0 ? String(row[tcCol] || '').toLowerCase().trim() : '';
    var status = statusCol >= 0 ? String(row[statusCol] || '').toLowerCase().trim() : '';

    if (VALID_TC.indexOf(tc) < 0 || VALID_STATUS.indexOf(status) < 0) continue;

    var qty = qtyCol >= 0 ? (Number(row[qtyCol]) || 1) : 1;
    total += qty;
    if (tc === 'vip') vip += qty; else std += qty;
    if (status === 'ql') ql += qty;
    if (row.join(' ').toLowerCase().indexOf('thai') >= 0) thai += qty;

    if (dateCol >= 0 && row[dateCol]) {
      var d = parseDate(row[dateCol]);
      if (d) {
        var yr = d.getFullYear();
        var startOfYear = new Date(yr, 0, 1);
        var wk = Math.ceil((((d - startOfYear) / 86400000) + startOfYear.getDay() + 1) / 7);
        var key = 'W' + ('0' + wk).slice(-2);
        weekly[key] = (weekly[key] || 0) + qty;
      }
    }
  }

  return { total: total, vip: vip, std: std, ql: ql, thai: thai, weekly: weekly };
}

function findHeaderRow(data) {
  for (var i = 0; i < Math.min(data.length, 10); i++) {
    var text = data[i].join(' ').toLowerCase();
    if (
      text.indexOf('lead source') >= 0 ||
      text.indexOf('email') >= 0 ||
      text.indexOf('full name') >= 0 ||
      text.indexOf('ticket') >= 0 ||
      text.indexOf('participant') >= 0
    ) return i;
  }
  return 0;
}

function findCol(headers, names) {
  for (var n = 0; n < names.length; n++) {
    for (var i = 0; i < headers.length; i++) {
      if (headers[i] === names[n] || headers[i].indexOf(names[n]) >= 0) return i;
    }
  }
  return -1;
}

function parseDate(value) {
  try {
    if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) return value;
    if (typeof value === 'number') return new Date(Math.round((value - 25569) * 86400000));
    var d = new Date(value);
    if (!isNaN(d.getTime())) return d;
  } catch (err) {}
  return null;
}

function outputJson(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
