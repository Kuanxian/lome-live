function lomedoGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tab = ss.getSheetByName("Full List") || ss.getSheets()[0];
  var data = tab.getDataRange().getValues();
  if(data.length < 2) return ContentService.createTextOutput('{"total":0}').setMimeType(ContentService.MimeType.JSON);
  var headers = data[0].map(function(h){ return String(h).toLowerCase().trim(); });
  var lsCol=-1,pgCol=-1,dateCol=-1;
  for(var c=0;c=0) lsCol=c;
    if(headers[c].indexOf("product group")>=0) pgCol=c;
    if(headers[c].indexOf("date")>=0&&dateCol<0) dateCol=c;
  }

  for(var i=1;i=0?String(row[lsCol]||"").trim():"";

    if(ls.toLowerCase().indexOf("refund")>=0) continue;
    if(lsSamples.length<5&&ls) lsSamples.push(ls);
    total++;
    var lsl=ls.toLowerCase(),pgl=pg.toLowerCase();
    if(lsl.indexOf("vip")>=0||pgl.indexOf("vip")>=0) vip++; else std++;
    if(lsl.indexOf("[ql]")===0||lsl.indexOf("ql buyer")>=0) ql++;
    if(dateCol>=0&&row[dateCol]){
      try{var d=new Date(row[dateCol]);if(!isNaN(d.getTime())){var yr=d.getFullYear();var wk=Math.ceil((((d-new Date(yr,0,1))/86400000)+new Date(yr,0,1).getDay()+1)/7);var key=yr+"-W"+(wk<10?"0"+wk:String(wk));weekly[key]=(weekly[key]||0)+1;}}catch(ex){}

  var result={total:total,vip:vip,std:std,ql:ql,vipPct:total>0?Math.round(vip/total*100):38,weekly:weekly,sheet:tab.getName(),totalRows:data.length-1,lsSamples:lsSamples,updated:new Date().toISOString()};
